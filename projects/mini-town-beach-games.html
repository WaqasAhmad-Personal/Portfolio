Production C# Implementation Pattern 

// Expression and Mood State Swapper for Doll Characters
public class DollCharacter : MonoBehaviour {
    [SerializeField] private SpriteRenderer mouthRenderer;
    [SerializeField] private Sprite happyMouth, eatingMouth, surprisedMouth;

    public void GiveFood(FoodItem food) {
        mouthRenderer.sprite = eatingMouth;
        // Trigger chewing animation tween
        Invoke(nameof(ResetMouth), 1.5f);
    }
    private void ResetMouth() => mouthRenderer.sprite = happyMouth;
}
