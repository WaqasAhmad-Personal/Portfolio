Production C# Implementation Pattern 

// Turn-Based Combat Action Executor
public class CombatTurnSystem : MonoBehaviour {
    public enum TurnState { PlayerTurn, EnemyTurn, Resolving }
    public TurnState currentState;

    public void ExecuteAttack(AttackData attack, Fighter target) {
        currentState = TurnState.Resolving;
        target.TakeDamage(attack.damage);
        // Play hit animation and particle effect
        Invoke(nameof(SwitchTurn), 1.2f);
    }

    private void SwitchTurn() {
        currentState = currentState == TurnState.PlayerTurn ? TurnState.EnemyTurn : TurnState.PlayerTurn;
    }
}
