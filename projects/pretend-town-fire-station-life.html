Production C# Implementation Pattern 

// Modular Item Pick-and-Place with Dynamic Depth Layering
public class InteractiveProp : MonoBehaviour {
    private SpriteRenderer _renderer;
    private int _originalSortOrder;

    private void Awake() {
        _renderer = GetComponent<SpriteRenderer>();
        _originalSortOrder = _renderer.sortingOrder;
    }

    public void OnPickup() => _renderer.sortingOrder = 999; // Float to top while dragging
    public void OnDrop(Transform slot) {
        transform.position = slot ? slot.position : transform.position;
        _renderer.sortingOrder = _originalSortOrder;
    }
}
