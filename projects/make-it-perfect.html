// Plane-Projected Drag and Magnetic Snap Mechanism
public class SnapDraggable : MonoBehaviour {
    private Plane _dragPlane;
    private Camera _cam;
    [SerializeField] private Transform targetSlot;
    [SerializeField] private float snapDistance = 0.35f;

    private void OnMouseDown() {
        _cam = Camera.main;
        _dragPlane = new Plane(Vector3.up, transform.position);
    }

    private void OnMouseDrag() {
        Ray ray = _cam.ScreenPointToRay(Input.mousePosition);
        if (_dragPlane.Raycast(ray, out float enter)) {
            Vector3 hitPoint = ray.GetPoint(enter);
            transform.position = Vector3.Distance(hitPoint, targetSlot.position) < snapDistance 
                ? targetSlot.position 
                : hitPoint;
        }
    }
}
