Production C# Implementation Pattern 

// Refresh-Rate Agnostic Tap Jump Impulse Controller
public class TapShotBall : MonoBehaviour {
    [SerializeField] private Vector2 tapImpulse = new Vector2(2.5f, 6.2f);
    private Rigidbody2D _rb2d;

    private void Awake() => _rb2d = GetComponent<Rigidbody2D>();

    public void OnTapInput() {
        // Zero out downward velocity before applying impulse for crisp response
        _rb2d.linearVelocity = new Vector2(_rb2d.linearVelocity.x, 0);
        _rb2d.AddForce(tapImpulse, ForceMode2D.Impulse);
    }
}
