Production C# Implementation Pattern 

// High-Impact Bumper Impulse Resolver with Directional Haptics
public class BumperCollisionResolver : MonoBehaviour {
    [SerializeField] private float bounceMultiplier = 1.8f;
    private Rigidbody _rb;

    private void Awake() => _rb = GetComponent<Rigidbody>();

    private void OnCollisionEnter(Collision col) {
        if (!col.gameObject.CompareTag("BumperCar")) return;
        Vector3 normal = col.contacts[0].normal;
        float relativeSpeed = col.relativeVelocity.magnitude;
        _rb.AddForce(-normal * (relativeSpeed * bounceMultiplier), ForceMode.Impulse);
    }
}
