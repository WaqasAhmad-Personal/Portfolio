Production C# Implementation Pattern 

// Zero-Allocation Recursive Bubble Splitter
public class BubbleEntity : MonoBehaviour {
    public int tier = 4; // 4: Giant, 3: Large, 2: Medium, 1: Small
    [SerializeField] private BubbleEntity childPrefab;

    public void SplitAndPop() {
        if (tier > 1) {
            SpawnChild(Vector2.left);
            SpawnChild(Vector2.right);
        }
        BubblePool.Instance.ReturnToPool(this);
    }

    private void SpawnChild(Vector2 direction) {
        var child = BubblePool.Instance.GetBubble(tier - 1);
        child.transform.position = transform.position;
        child.Launch(direction);
    }
}

🚀 Profiling & Hardware Optimization

Pre-allocated bubble object pool for all 4 size tiers; zero Instantiate calls during gameplay.
